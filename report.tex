\abstract{
In this work, we detail the effect of continuous symmetries on classical gauge theories
namely, Noether's theorems, before gauging the Dirac Lagrangian's 
axial symmetry to obtain a quantum electrodynamical analogue of the
interaction between massless Dirac fermions and a $\mathrm{U}(1)$ axial boson;
an `axial theory'.
Subsequently, we investigate the effect of a classical theory's symmetries
on the quantum field theory (QFT) via Slavnov-Taylor and Ward-Takahashi identities,
and we find that, for the class of symmetries with infinitesimal generators
linear in the dynamical fields, a singular path integral Jacobian matrix
characterises the presence of an anomaly in the quantised theory. Moreover,
anomalies can be seen as topological invariants of a QFT, and gauging an
anomalous `linear' symmetry results in a violation of unitarity;
an inconsistent quantum theory. In the case of the `axial theory',
we compute the anomaly explicitly.}

\section{Introduction}

The study of continuous symmetries in dynamical systems has long been
used by theorists to constrain possible Lagrangians. To this end, the seminal
results in classical field theory are Noether's first and second theorems\cite{Noether1918}, 
which reveal the correspondence between quantities conserved on-shell and 
infinitesimal symmetries. Moreover, the analysis of local (gauge) symmetries
has produced incredibly successful gauge theories such as quantum electrodynamics\cite{Schwinger},
which can be seen as resultant from `gauged' symmetries. In this vein, we show that gauging the
axial symmetry of the Dirac Lagrangian results in a 
well-formulated classical gauge theory; an `axial theory' with a 
$\mathrm{U}(1)$ axial boson.

Thereafter, we investigate the effect of symmetries of the classical action
on QFT via the quantum field-theoretic analogue of Noether's theorems; the
Slavnov–Taylor and Ward-Takahashi identities~\cite{Slavnov,Taylor,Slavnov–Taylor,WardWTI,TakahashiWTI}.
Anomalous symmetries are symmetries of the classical theory that are violated by quantum processes in
a QFT\@. In particular, the axial symmetry is famously violated
by triangle feynman diagrams (the Adler-Bell-Jackiw anomaly~\cite{Adler,Bell-Jackiw}) and can be 
viewed as a topological invariant of QFT via the Atiyah-Singer index
theorem~\cite{Atiyah-Singer}. Using the Slavnov–Taylor identities, we find that it is 
necessary and sufficient for an anomalous symmetry
with infinitesimal generators linear in the dynamical fields to have
a singular path integral Jacobian, and gauging such symmetries
produces a gauge-variant partition function; an inconsistent quantum theory.

Finally, we compute the gauge-variance of the `axial theory' partition function
using Fujikawa's method~\cite{Fujikawa}(although a topological
approach is possible~\cite{AlvarezGuameNon-AB}),
demonstrating an inconsistency in the quantum theory and thus
a $\mathrm{U}(1)$ axial boson coupled to Dirac fermions does not exist.

\section{Symmetries in Classical Gauge theory}

\subsection{Classical field theory}
In classical mechanics we consider a countable set of particles
each with finitely many degrees of freedom
and generalised coordinates $q_i(t)$. These generalised coordinates ${\{ q_i(t) \}}_i$ specify
the system's configuration (position in configuration space) and together with the 
generalised conjugate momenta
${\left\{p_i(t) \coloneq \frac{\partial L}{\partial \dot{q}_i} \right\}}_i $ 
define the system's position in phase space~\cite{LA&LI-Mech-I}.
In classical field theory we generalise this notion of configuration space to a
continuum with infinite degrees of freedom. The scalar field $\phi(t)$ can be seen as
the generalised coordinates of a continuum 
$\left( q_i(t) \xrightarrow{i \rightarrow \vec{x}} \phi(\vec{x}, t) \right)$ and for a system
of continua the set of scalar fields ${\{ \phi_k(\vec{x}, t) \}}_k$
specifies the system's configuration~\cite{ClassField}.
Subsequently, we generalise the classical Lagrangian $L(q_i(t), \dot{q}_i(t), t)$ via
\begin{equation}
  L(t) = \int{d^{3}\vec{x}\,
  \mathcal{L}(\phi_k(\vec{x}, t),\partial_{\mu}\phi_k(\vec{x}, t), \vec{x}, t)},
\end{equation}
where $\mathcal{L}$ is the Lagrangian density. With respect to a path
in configuration space, $\vec{\Phi}(\vec{x}, t)$, the classical action becomes 
\begin{equation}
  S\left[\vec{\Phi}(\vec{x}, t) \right] = \int{dt\,L} 
  = \int{d^4x\,\mathcal{L}(\phi_k(\vec{x}, t),\partial_{\mu}\phi_k(\vec{x}, t), \vec{x}, t)}.
\end{equation}
Using Hamilton's principle~\cite{LA&LI-Mech-VII} ($\delta S[\vec{\Phi}; \vec{\rho} ] = 0,
\forall\vec{\rho} \in C_{0}^{\infty})$\footnote{$\delta{S}$ is a Gateaux derivative,
and $\vec{\rho}$ vanishes at the endpoints of $\vec{\Phi}$}
we obtain the field-theoretic Euler-Lagrange equations 
\begin{equation}
  \frac{\partial \mathcal{L}}{\partial \phi_k}
  \approx \partial_\mu \left[ \frac{\partial \mathcal{L}}{\partial(\partial_\mu{\phi_k}) } \right],
\end{equation}
where we use $\approx$ to denote an on-shell equality.
In Lagrangian systems the Hamiltonian, $H(p_i(t), q_i(t), t)$,
is characterised by the Legendre transform
\begin{equation}
  H = \sum_{i}{p_i(t)\dot{q}_i(t) - L},
\end{equation}
to generalise this to field theory we introduce the momentum field conjugate to 
$\phi_k(\vec{x}, t)$
\begin{equation}
  \pi_k(\vec{x}, t) \coloneq \frac{\partial \mathcal{L} }{\partial \dot{\phi}_k},
\end{equation}
\begin{equation}
  p_i(t) \coloneq \frac{\partial L}{\partial \dot{q}_i} \xrightarrow{i \rightarrow \vec{x}}
  \frac{\partial}{\partial \dot{\phi}_k} \left[ \int{d^3\vec{x}\,\mathcal{L}} \right]
  = \int{d^3\vec{x}\,\pi_k(\vec{x}, t)}.
\end{equation}
Thus the classical Hamiltonian is generalised to
\begin{equation}
  H(t) = \int{d^3\vec{x}\,\mathcal{H}(\pi_k(\vec{x}, t), \phi_k(\vec{x}, t), \vec{x}, t) },
\end{equation}
\begin{equation}
  \mathcal{H} = \sum_{k}{\pi_k(\vec{x}, t)\dot{\phi}_k(\vec{x}, t) - \mathcal{L}},
\end{equation}
where $\mathcal{H}$ is the Hamiltonian density, and the field-theoretic 
analogue of Hamilton's equations~\cite{LA&LI-Mech-VII} is
the Hamiltonian field equations~\cite{ClassField}
\begin{equation}
  \dot{\phi}_k = \frac{\delta \mathcal{H}}{\delta \pi_k}, \quad
  \dot{\pi}_k = - \frac{\delta \mathcal{H}}{\delta \phi_k},
\end{equation}
where we use the functional derivative identify 
\begin{equation}
  \frac{\delta \mathcal{F}}{\delta g} = \frac{\partial \mathcal{F}}{\partial g}
  - \partial_\mu \left[ \frac{\partial \mathcal{F}}{\partial (\partial_\mu g)} \right].
\end{equation}
To begin second quantisation we will also require the notion of a field-theoretic poisson bracket.
Given two functionals, $F$ and $G$, of the dynamical
fields given by\footnote{n.b. $K({x}) = \int{d{y}\,[K({y}) \delta(x - y)]}$}  
\begin{equation}
  F = \int{d^3\vec{x}\, \mathcal{F}(\pi_k, \phi_k, \vec{x}, t)}, \quad
  G = \int{d^3\vec{x}\, \mathcal{G}(\pi_k, \phi_k, \vec{x}, t)},
\end{equation}
we define the poisson bracket in field theory via
\begin{equation}
  {\{F, G \}}_{f} = \int{d^3\vec{x}\,
  \sum_{k}{\left[ \frac{\delta \mathcal{F}}{\delta \phi_k}\frac{\delta \mathcal{G}}{\delta \pi_k} 
- \frac{\delta \mathcal{G}}{\delta \phi_k}\frac{\delta \mathcal{F}}{\delta \pi_k} \right]}  }.
\end{equation}
 

\subsection{Noether's first theorem}
In classical mechanics Noether's first theorem shows the correspondence between global symmetries of 
the lagrangian and conserved quantities called Noether charges. This generalises to global symmetries
of the lagrangian density corresponding to conserved Noether currents in classical field theory.
Consider an infinitesimal field transformation $\vartheta(\epsilon)$ such that
\begin{equation}
  \phi_n \mapsto \phi_n + \delta_\epsilon\phi_n + \mathcal{O}(\epsilon^2); \quad 
  \delta_\epsilon\phi_n = \epsilon\vartheta_n(\phi_k),
\end{equation}
where the generators, $\vartheta_k$, are independent of spacetime.
A transformation is called a global symmetry if its effect on the Lagrangian density is
\begin{equation}
  \mathcal{L} \mapsto \mathcal{L} + \epsilon\partial_\mu\Lambda^\mu(\phi_k, x)
  + \mathcal{O}(\epsilon^2).
\end{equation}
This transformation changes the action by a surface term and the Euler-Lagrange equations
are invariant. Using a Taylor expansion of the Lagrangian density under $\vartheta(\epsilon)$
we find that
\begin{equation}\label{eq:6}
  \delta_\epsilon\mathcal{L} = \sum_k{\delta_\epsilon\phi_k E_k} + \partial_\mu\theta^\mu,
\end{equation}
\begin{equation}
  \quad \theta^\mu = \sum_k{\delta_\epsilon\phi_k
  \frac{\partial \mathcal{L}}{\partial(\partial_\mu{\phi_k}) } }.
\end{equation}
Where $E_k$ are the Euler-Lagrange derivatives and $\theta^\mu$ is
the symplectic potential density~\cite{LocalSymm}.
By considering the shift in the action we arrive at
Noether's first theorem~\cite{Noether1918}; an 
infinitesimal transformation $\vartheta(\epsilon)$ is a symmetry
of the Lagrangian if and only if 
\begin{equation}\label{eq:1}
  \epsilon j^\mu \coloneq \theta^\mu - \epsilon\Lambda^\mu,
  \quad \epsilon\partial_\mu j^\mu = - \delta_\epsilon\phi_k E_k \approx 0.
\end{equation}
Where we use the summation convention for the field indices hereafter
and the on-shell divergence-free quantity $j^\mu$ is called the conserved
Noether current\footnote{n.b. $j^\mu$ is $\epsilon$-independent}.
This easily generalises to local symmetries: spacetime dependent generators
$\vartheta_k(\phi, x)$. Moreover, a Lagrangian with no
explicit spacetime dependence implies that the canonical energy-momentum
tensor obeys divergence relations on-shell 
\begin{equation}
  {T^\mu}_\nu = \frac{\partial \mathcal{L}}{\partial(\partial_\mu{\phi_k})} 
  \partial_\nu\phi_k -\delta^\mu_\nu\mathcal{L},
  \quad \partial_\mu {T^\mu}_\nu \approx 0,
\end{equation}
which can be seen as the independent conservation relations of
four Noether currents. By Noether's first theorem the transformations corresponding to
these Noether currents are the global symmetries
\begin{equation}
  \phi_n(x^\mu) \mapsto \phi_n(x^\mu + \epsilon) = \phi_n(x^\mu)
  + \epsilon\partial_\mu\phi_n\big{|}_{x^\mu} + \mathcal{O}(\epsilon^2), 
\end{equation}
and it is clear that translation invariance of the equations of motion (for spacetime independent 
Lagrangians) is a direct corollary of Noether's first theorem.

\subsection{Noether's second theorem}
Noether's second theorem shows that a local symmetry introduces off-shell constraints
on the equations of motion and leads to Noether current identities which are
crucial for gauge theories. Under
local infinitesimal symmetries (gauge symmetries), $R_\xi$, of the form
\begin{equation}\label{eq:5}
   \phi_n \mapsto \phi_n + \delta_\xi\phi_n + \mathcal{O}(\xi^2);
   \quad \delta_\xi\phi_n = \sum_{m=0}{R^{\mu_1 \cdots \mu_m}_n(\phi_k)
   \partial_{\mu_1}\cdots\partial_{\mu_m} \xi\lambda(x)},
\end{equation}
the action shifts by
\begin{equation}
  \delta_\xi S = \int{d^4x\, \delta_\xi\phi_n E_n + \partial_\mu\theta^\mu}.
\end{equation}
If $\lambda(x)$ is a compactly supported smooth function, the boundary term vanishes and we have
\begin{equation}
  \int{d^4x\, \delta_\xi\phi_n E_n} = 0,
\end{equation}
and integration by parts leads to 
\begin{equation}
  \int{d^4x\, \lambda(x)\Delta(E) } = 0; \quad
  \Delta(E) \coloneq \sum_{m=0}{ {(-1)}^m \partial_{\mu_1}\cdots\partial_{\mu_m}
  [R^{\mu_1 \cdots \mu_m}_n(\phi_k) E_n] }. 
\end{equation}
By the fundamental lemma of the calculus of variations~\cite{calcvar} we obtain the 
off-shell constraint $\Delta(E) = 0$. Thus the equations of motion are not
all independent and the Euler-Lagrange equations ($E_n = 0$)
are under-determined. However, observables must be uniquely determined
by the theory hence we conclude some degrees of freedom are gauge. Using
the surface term $S^\mu$ from integration by parts we find an expression similar to~\eqref{eq:1}
\begin{equation}
 \delta_\xi\phi_n E_n = \xi\partial_\mu S^\mu(E_k, \lambda),
\end{equation}
with the important difference that $S^\mu$ vanishes on-shell. The
quantity $j^\mu + S^\mu$ is divergence-free off-shell, subsequently
by the Poincaré lemma we obtain
Noether's second theorem~\cite{Noether1918, Noeth2Ward, SuperPot};
on-shell the canonical Noether current $j^\mu$ is
the divergence of a rank-2 tensor
\begin{equation}\label{eq:2}
  j^\mu = -S^\mu(E_k, \lambda) + \partial_\nu \kappa^{\mu\nu}
  \approx \partial_\nu \kappa^{\mu\nu};
  \quad \kappa^{\mu\nu} = -\kappa^{\nu\mu},
\end{equation}
the antisymmetric tensor $\kappa^{\mu\nu}$ is called a superpotential.
Although the superpotential is arbitrary ($\kappa^{\mu\nu}$
is determined up to addition by a divergence-free quantity) 
it can be uniquely determined if
one demands the superpotential obeys additional constraints
(such as asymptotic conservation)~\cite{SuperPot}. In the case of electrodynamics
the superpotential is electromagnetic field tensor and~\eqref{eq:2}
becomes Maxwell's equations
\begin{equation}
  J^\mu \approx \partial_\nu F^{\nu\mu},
  \quad F_{\mu\nu} = \partial_{\mu}A_{\nu} - \partial_{\nu}A_{\mu}.
\end{equation}


\section{The Dirac Lagrangian}

\subsection{Gauging the vector symmetry}
In quantum mechanics the Schrodinger equation is not Lorentz invariant and is thus
incompatible with special relativity. A generalisation of the relativistic
dispersion relation using Hamiltonian and momentum operators
for free particles leads to the Klein-Gordon
equation\footnote{$(\square + m^2)\psi = 0$}~\cite{Klein1926, Gordon1926}.
However, the Klein-Gordon equation is a second order partial differential equation and
subsequently does not uniquely determine time-evolution of the wavefunction.
Alternatively, by finding the half-iterate of d'Alembert
operator\footnote{$\square \coloneq \partial^2_t - \nabla^2$}~the 
Klein-Gordon equation simplifies to a first order partial differential equation;
the Dirac equation~\cite{dirac1958}
\begin{equation}
  (i \gamma^\mu\partial_\mu - m)\psi(x) = 0.
\end{equation}
We call the corresponding Lagrangian density the Dirac Lagrangian
\begin{equation}
  \mathcal{L}_\mathrm{D} = \bar{\psi}(i \gamma^\mu\partial_\mu - m)\psi,
\end{equation}
where the adjoint bispinor is defined as $\bar{\psi} \coloneq \psi^\dagger\gamma^0$.
This Lagrangian admits a global $\mathrm{U}(1)$ symmetry called the vector symmetry
\begin{equation}
  \psi \mapsto e^{i\vartheta}\psi, \quad \bar{\psi} \mapsto \bar{\psi}e^{-i\vartheta}.
\end{equation}
This symmetry corresponds to a conserved current, called the vector current,
via Noether's first theorem
\begin{equation}
  j^\mu = \bar{\psi}\gamma^\mu\psi.
\end{equation}
We now attempt to `gauge' this symmetry by adding spacetime dependence
to the infinitesimal parameter
\begin{equation}
  \psi \mapsto e^{i\vartheta(x)}\psi, \quad \bar{\psi} \mapsto \bar{\psi}e^{-i\vartheta(x)}.
\end{equation}
However, the Lagrangian is not invariant under such a transformation so we introduce a
derivative operator which transforms covariantly
\begin{equation}\label{eq:3}
  D_\mu \coloneq \partial_\mu + ie\Pi_\mu,
\end{equation}
\begin{equation}
  D_\mu \psi \mapsto D'_\mu\left[e^{i\vartheta(x)}\psi\right] = e^{i\vartheta(x)}D_\mu\psi.
\end{equation}
To satisfy this the gauge field must transform as
\begin{equation}
  \Pi_\mu \mapsto \Pi_\mu -\frac{1}{e}\partial_\mu\vartheta(x),
\end{equation}
Thus we have a modified Lagrangian which admits a local $\mathrm{U}(1)$ gauge symmetry
\begin{equation}
  \mathcal{L} = \bar{\psi}(i\slashed{D} - m)\psi
  = \bar{\psi}(i \gamma^\mu\partial_\mu - m)\psi -e\Pi_\mu j^\mu,
\end{equation}
where $\slashed{D} \coloneq \gamma^\mu D_\mu$ is the Dirac operator.
This is strikingly similar to the electromagnetic Lagrangian density which
also has a $\mathrm{U}(1)$ gauge symmetry
\begin{equation}
  \mathcal{L}_\mathrm{EM} = -\frac{1}{4}F_{\mu\nu}F^{\mu\nu} -A_\mu J^\mu.
\end{equation}
Thus we identify $\Pi_\mu = A_\mu$ and $ej^\mu = J^\mu$
and arrive at the Lagrangian for quantum electrodynamics.
\begin{equation}
  \mathcal{L}_\mathrm{QED} = -\frac{1}{4}F_{\mu\nu}F^{\mu\nu} -eA_\mu j^\mu
    + \bar{\psi}(i \gamma^\mu\partial_\mu - m)\psi.
\end{equation}
Schwinger arrived at the covariant formulation of quantum electrodynamics
similarly by constructing a Lagrangian invariant under Lorentz transformations,
gauge transformations, and charge conjugation~\cite{Schwinger}. 
However, this is not the end of the story, the massless Dirac equation
admits another global $\mathrm{U}(1)$ symmetry. In the next sections 
we construct a gauge-invariant Lagrangian and investigate
the resultant gauge theory.

\subsection{Gauging the axial symmetry}

The massless Dirac Lagrangian admits
another symmetry better seen in the Weyl basis
\begin{equation}
  \psi =
  \begin{pmatrix} \psi_\mathrm{L} \\ \psi_\mathrm{R} \end{pmatrix}, \quad
  \gamma^{\mu}\partial_\mu =
  \begin{pmatrix}0 & \sigma^\mu\partial_\mu \\ \bar{\sigma}^{\mu}\partial_\mu & 0 \end{pmatrix}.
\end{equation}
In this basis the Dirac Lagrangian can be seen as the interaction between two Weyl fermions
of opposite chirality~\cite{Weyl}
\begin{equation}
  \mathcal{L}_\mathrm{D} = \psi_\mathrm{L}^{\dagger}(i\sigma^\mu\partial_\mu)\psi_\mathrm{L} +
  \psi_\mathrm{R}^{\dagger}(i\bar{\sigma}^\mu\partial_\mu)\psi_\mathrm{R}
  - m(\psi_\mathrm{L}^{\dagger}\psi_\mathrm{R} + \psi_\mathrm{R}^{\dagger}\psi_\mathrm{L}).
\end{equation}
In the massless case the Lagrangian gains a global $\mathrm{U}(1)$ symmetry
known as the axial symmetry
\begin{equation}
  \psi_\mathrm{L} \mapsto e^{i\vartheta}\psi_\mathrm{L}, \quad
  \psi_\mathrm{R} \mapsto e^{-i\vartheta}\psi_\mathrm{R},
\end{equation}
which is equivalent to 
\begin{equation}
  \psi \mapsto e^{i\vartheta\gamma^5}\psi,
\end{equation}
and a Noether current
\begin{equation}
  j^\mu_5 = \bar{\psi}\gamma^\mu\gamma^5\psi, \quad \partial_\mu j^\mu_5 = 2im\bar{\psi}\gamma^5\psi.
\end{equation}
We now proceed à la Schwinger, gauging the axial symmetry by promoting $\partial_\mu$
to a covariant derivative $\widetilde{D}_\mu$
as in~\eqref{eq:3} (with $q=e$) and determining the required gauge transformation
\begin{equation}
  {\mathcal{L}_{D}|}_{m=0} = \bar{\psi}(i\gamma^\mu \partial_\mu)\psi
  - q\bar{\psi}\gamma^\mu\Pi_\mu\psi,
\end{equation}
\begin{equation}
  \Pi_\mu \mapsto e^{i\vartheta\gamma^5}\Pi_\mu e^{-i\vartheta\gamma^5} 
  - \frac{1}{q}\partial_\mu\vartheta\gamma^5.
\end{equation}
Now we must add a kinetic term to the Lagrangian to determine the dynamics 
of the matrix-valued gauge field\footnote{n.b. $\Pi^{ab}_\mu$ has two
internal indices and one spacetime index}.
Such a term must be gauge-invariant to preserve
the $\mathrm{U}(1)$ symmetry. We find a Lagrangian analogous to the Maxwell Lagrangian
\begin{equation}
  \mathcal{L}_\Pi = - \frac{1}{4}\mathrm{Tr}({K_{\mu\nu}K^{\mu\nu}}),
\end{equation}
where the trace is taken over the internal indices and
\begin{equation}
  K_{\mu\nu} = \partial_\mu\Pi_\nu - \partial_\nu\Pi_\mu +iq[\Pi_\mu, \Pi_\nu],
\end{equation}
has the required gauge-invariance and is Lorentz invariant.
Thus we have constructed a classical gauge theory of massless Dirac fermions
\begin{equation}
  \mathcal{L}_\mathrm{A} =  - \frac{1}{4}\mathrm{Tr}({K_{\mu\nu}K^{\mu\nu}}) 
  + \bar{\psi}i\widetilde{\slashed{D}}\psi,
\end{equation}
\begin{equation}\label{eq:9}
  \mathcal{L}_\mathrm{A} =  - \frac{1}{4}\mathrm{Tr}({K_{\mu\nu}K^{\mu\nu}})
  - q\bar{\psi}\gamma^\mu\Pi_\mu\psi
  + \bar{\psi}(i\gamma^\mu \partial_\mu)\psi.
\end{equation}
By direct computation we find that the canonical energy-momentum tensor for this theory is
\begin{equation}
  {T^\mu}_\nu = i\bar{\psi}\gamma^\mu\partial_\nu\psi
  + \mathrm{Tr}(K^{\kappa\mu}\partial_\nu\Pi_\kappa) - \delta^\mu_\nu \mathcal{L}_\mathrm{A},
\end{equation}
which is not gauge-invariant. However, using the procedure for interacting gauge theories
in~\cite{energy–momentum-tensor} we construct an improved energy-momentum tensor
\begin{equation}
 \widetilde{T}^\mu{}_\nu = i\bar{\psi}\gamma^\mu\widetilde{D}_\nu\psi
 + \mathrm{Tr}(K_{\kappa\mu}K^{\nu\kappa}) - \delta^\mu_\nu\mathcal{L}_\mathrm{A},
\end{equation}
which is clearly gauge-invariant. Thus gauging the axial symmetry results in a 
well formulated classical gauge theory of an axial boson and massless Dirac
fermions.

\section{Quantum Field Theory}

So far we have studied relativistic quantum mechanics, making use of classical
fields to describe the dynamics of quantum states. However, such a quantum theory is
inconsistent and predicts particles with negative energy which require 
peculiar interpretations to reconcile like Dirac's infamous hole theory~\cite{Holethry}.
To proceed to a quantum theory of particle interactions we must promote the dynamical fields
to field operators; second quantisation. We shall see how this shift to
a quantised field theory affects our calculation of expectation values and probabilities
via the path integral.

\subsection{Second Quantisation}
In classical mechanics, the canonical transformations are exactly those which preserve the symplectic 
structure; invariance of poisson brackets of the dynamical variables ($p_i, q_i$) characterises
transformations which leave Hamilton's equations unchanged~\cite{LA&LI-Mech-VII}.
A system's position in phase space specifies its classical state.
However, in quantum mechanics all properties of a system are included in a quantum state, $\ket{\psi}$, 
inside of a Hilbert space upon which operators corresponding to observables act.

Dirac's famous canonical quantisation
rule (${\{A, B \} \rightarrow \frac{1}{i\hbar}[\hat{A}, \hat{B}]}$)~\cite{dirac1958}
allows us quantise the canonical structure of
classical mechanics (although Groenewold's theorem shows us 
that such a rule cannot hold for all functions of the dynamical variables~\cite{GROENEWOLD}). 
To obtain a quantised field theory we consider the field-theoretic poisson brackets
\begin{equation}
  {\{ \phi_n(z), \phi_m(w) \}}_f = 0 = {\{ \pi_n(z), \pi_m(w) \}}_f,
\end{equation}
\begin{equation}
  {\{\phi_n(z), \pi_m(w) \}}_f = \delta^n_m\delta^{(3)}(\vec{z} - \vec{w}).
\end{equation}
Which are quantised to the canonical commutation relations for the dynamical
field operators acting on a Fock space~\cite{FockSpace}
\begin{equation}
  {[\hat{\phi}_n(z), \hat{\phi}_m(w) ]} = 0
  = {[\hat{\pi}_n(z), \hat{\pi}_m(w) ]},
\end{equation}
\begin{equation}
  {[ \hat{\phi}_n(z), \hat{\pi}_m(w) ]}
  = i\hbar\delta^n_m\delta^{(3)}(\vec{z} - \vec{w}),
\end{equation}
where the classical field $\phi(z)$ is an eigenvalue of
the operator $\hat{\phi}(z)$.
However, just like Dirac's rule this procedure does not always produce a consistent quantum
theory. In the case of fermionic fields canonical quantisation
produces states with negative probability and a Hamiltonian 
unbounded below~\cite{NtveEnrgy}; we require anti-commutation
relations (because of the spin-statistics theorem~\cite{SpinStat})
in natural units\footnote{n.b. the momentum conjugate to $\psi_\alpha$ is 
$i\psi^\dagger_\alpha$}
\begin{equation}
  {[\hat{\psi}_\alpha(z), \hat{\psi}_\beta^\dagger(w)]}_{+}
  = \delta^\alpha_\beta\delta^{(3)}(\vec{z} - \vec{w}),
\end{equation}
where the unwritten anticommutators vanish.
This implies the Dirac spinors are Grassmann (anticommuting)
variables\footnote{n.b. $\phi^2$ is an eigenvalue of $\hat{\phi}^2 = 0$}
and we require Berezin's notion of calculus on a Grassmann algebra.

\subsection{The path integral}
In a quantised field theory the path integral can be used to
express expectation values of operators and their compositions
\begin{equation}
  \int{\mathcal{D}\phi\,\left\{\varphi_1(x_1) \cdots \varphi_n(x_n)\right\}e^{iS[\phi]}} \propto
  \braket{{\varphi_1(x_1) \cdots \varphi_n(x_n)}} \coloneq
  \braket{0|\mathcal{T}\{\hat{\varphi}_1(x_1) \cdots \hat{\varphi}_n(x_n)\}|0},
\end{equation}
where $\mathcal{T}$ is the time-ordering operator and we normalise probability
amplitudes using the partition function
\begin{equation}
  Z[J] = \int{\mathcal{D}\phi\,\exp{\left[iS[\phi] + i\int{d^4x\, J(x)\phi(x) }\right]} },
\end{equation}
\begin{equation}
  \frac{1}{Z[0]}\int{\mathcal{D}\phi\,\varphi(x)e^{iS[\phi]}} = \braket{\varphi(x)},
\end{equation}
and define the `quantum action' via
\begin{equation}
  W[J] \coloneq -i\log{Z[J]}.
\end{equation}
To preform calculations with the path integral we must
perform a wick transformation so that the Lagrangian is integrated 
over Euclidean space, after which we transform the time axis 
back to $\mathbb{R}$ via analytic continuation. In Euclidean 
space we have
\begin{equation}
  Z[J] = \int{\mathcal{D}\phi\,\exp{\left[-S_E[\phi] +
  \int{d^4{x_E}\,J(x)\phi(x)}\right] }}, \quad Z[J] = e^{-W_E[J]}.
\end{equation}


\section{Symmetries in QFT}

\subsection{Slavnov-Taylor identities}
We now extend our investigation of symmetries from classical fields to a
quantised theory. Suppose the classical action admits a local symmetry
such that the path integral measure is invariant
\begin{equation}
  \phi_\mu \mapsto \phi_\mu + \delta_\epsilon\phi_\mu +
  \mathcal{O}(\epsilon^2); \quad \delta_\epsilon\phi_\mu = \epsilon\vartheta_\mu(\phi, x),
\end{equation}
\begin{equation}
  \mathcal{D}\phi \mapsto
  \mathcal{D}\phi\,\det\!{\left(\frac{\partial \phi'_{\mu}(x)}{\partial \phi_\nu(y)}\right)}
  = \mathcal{D}\phi
\end{equation}
where $B_{\mu\nu} = \frac{\partial \phi'_{\mu}(x)}{\partial \phi_\nu(y)}$ is the 
transformation Jacobian. Thus the partition function $Z[J]$ is invariant
and for a source $J^\mu$ corresponding to $\phi_\mu$, and 
in Euclidean space we can write
\begin{equation}
  Z[J] = \int{\mathcal{D}\phi\,\exp{\left[-S[\phi]
    + \int{d^4x\, J^\mu(x)(\phi_\mu(x) + \delta_\epsilon\phi_\mu
    +\mathcal{O}(\epsilon^2) )}\right]} },
\end{equation}
\begin{equation}
  Z[J] = \int{\mathcal{D}\phi\,\left\{ 1 + \int{d^4x\,J^\mu(x)\delta_\epsilon\phi_\mu
    + \mathcal{O}(\epsilon^2)} \right\}
  \exp{\left[-S[\phi] + \int{d^4\,J^\mu(x)\phi_\mu(x)}\right]} },
\end{equation}
and after interchanging the spacetime and path integrals we have
\begin{equation}\label{eq:4}
  \int{d^4x\,J^\mu(x)\!\braket{\vartheta_\mu(\phi, x)}}_J = 0.
\end{equation}
We define the effective quantum action as the Legendre transform
\begin{equation}
  \Gamma[\varphi] = W[J] - \int{d^4x\,J^\mu(x)\varphi_\mu(x)}; \quad 
  \quad \varphi_\mu \coloneq \braket{\phi_\mu}_J.
\end{equation}
Taking the functional derivative we obtain
\begin{equation}
  J^\mu(x) = - \frac{\delta \Gamma[\varphi]}{\delta \varphi_\mu(x)},
\end{equation}
and after using~\eqref{eq:4} and the definition
of a functional derivative we have
\begin{equation}
  \delta\Gamma[\varphi_\mu;\braket{\vartheta_\mu(\phi, x)}_J] = 0.
\end{equation}
Thus the effective quantum action is invariant under the transformations
\begin{equation}
  \varphi_\mu \mapsto \varphi_\mu + \epsilon\!\braket{\vartheta_\mu(\phi, x)}_J,
\end{equation}
such a relation is called a Slavnov-Taylor identity~\cite{Slavnov, Taylor, Slavnov–Taylor}.
If the transformation is linear in the
fields\footnote{i.e. linear $R^\mu_n(\phi_k)$ in~\eqref{eq:5}} such that
\begin{equation}\label{eq:7}
  \vartheta_\mu = \Theta_\mu[\phi, x] = \alpha_\mu(x) +
  \int{d^4y\,\beta^\nu_\mu(x, y)\phi_\nu(y) },
\end{equation}
the transformation Jacobian becomes field independent
\begin{equation}
  B_{\mu\nu} = \frac{\delta}{\delta\phi_\nu(y)}(\phi_\mu +\epsilon\Theta_\mu[\phi, x])
  = \delta_{\mu\nu}\delta^{(4)}(x - y) + \epsilon\beta^\nu_\mu(x, y),
\end{equation}
and can be brought out of the path integral.
Thus even if the path integral measure is not invariant 
the normalised correlators will be, granted the Jacobian is non-singular.
Moreover, we have
\begin{equation}
\braket{\Theta_\mu[\phi, x]}_J = \alpha_\mu(x) +
\int{d^4y\,\beta^\nu_\mu(x, y)\braket{\phi_\nu(y)}_J } = \Theta_\mu[\varphi, x],
\end{equation}
which implies the effective quantum action is
invariant under the same transformation as the 
classical action, namely,
\begin{equation}
  \varphi_\mu \mapsto \varphi_\mu + \epsilon\Theta_\mu[\varphi, x],
\end{equation}
is a symmetry of $\Gamma[\varphi]$. In this case, the
quantum theory inherits the classical symmetry
and it is impossible to violate the symmetry via quantum effects
like loop diagrams because the corresponding correlators are invariant~\cite{Slavnov–Taylor}.

\subsection{Ward-Takahashi identities}

The classical conservation laws for Noether currents present in classical field theory
have an analogue in the quantum theory, the Ward-Takahashi identities.
Noether's first theorem in the classical theory induces
conservation relations for correlators in the quantum field theory.
Consider a bounded functional of the fields $F[\phi]$ and its vacuum expectation value
\begin{equation}
  \braket{F[\phi]} = \frac{1}{Z[0]}\int{\mathcal{D}\phi'\,F[\phi']e^{iS[\phi']}},
\end{equation}
\begin{equation}
  \quad \phi' = \phi + \delta_\epsilon\phi +\mathcal{O}(\epsilon^2);
  \quad \delta_\epsilon\phi = \epsilon\lambda(\phi,x),
\end{equation}
which follows from relabeling the variable of integration. Suppose
the field transformation preserves the path integral measure, then
taking the functional taylor expansion leads to
\begin{equation}
  \braket{F[\phi]} = \frac{1}{Z[0]}\int{\mathcal{D}\phi\,\left(F[\phi] +
    \frac{\delta F[\phi]}{\delta\phi}\epsilon\lambda +\mathcal{O}(\epsilon^2) \right)
    \exp{\left[ iS{[\phi]} + i\frac{\delta S[\phi]}{\delta\phi}\epsilon\lambda
  + \mathcal{O}(\epsilon^2) \right]}},
\end{equation}
\begin{equation}
  \braket{F[\phi]} = \frac{1}{Z[0]}\int{\mathcal{D}\phi\,
    \left(F[\phi] + \frac{\delta F[\phi]}{\delta\phi}\epsilon\lambda \right)
    \left(1 + i\frac{\delta S[\phi]}{\delta\phi}\epsilon\lambda\right)e^{iS[\phi]} }
    + \mathcal{O}(\epsilon^2),
\end{equation}
and to first-order in $\epsilon$ we have
\begin{equation}
 \int{\mathcal{D}\phi\,\lambda(\phi,x) \left(
    \frac{\delta F[\phi]}{\delta\phi} +
    iF[\phi]\frac{\delta S[\phi]}{\delta\phi} \right)e^{iS[\phi]} } = 0.
\end{equation}
Since $\lambda(\phi,x)$ is smooth and arbitrary this implies the
Dyson-Schwinger equation~\cite{DysonDSE,SchwingerDSE}
\begin{equation}
  \Braket{\frac{\delta F[\phi]}{\delta\phi}} = -i\Braket{F[\phi]\frac{\delta S[\phi]}{\delta\phi}}.
\end{equation}
Suppose the classical Noether current $j^\mu(x)$ has an analogue $\hat{\jmath}^\mu(x)$
in the operator formalism and the operators
$\{\hat{\mathcal{V}}_i(z_i)\}$ are functions of spacetime. The 
derivative of their correlator is
\begin{multline}
  \partial^x_{\mu}\!\Braket{j^\mu(x)\prod^n_{i=1}{\!\mathcal{V}_i(z_i)}} = 
  \Braket{\partial^x_{\mu}j^\mu(x) \prod^n_{i=1}{\!\mathcal{V}_i(z_i)}} \\ + 
  \mathlarger{\mathlarger{\sum}}_{k = 1}^{n}{ \delta(x_0 - z_{k_0})
    \!\Braket{\left[j^0(x), \mathcal{V}_k(z_k)\right]\prod^n_{i\neq k}{\!\mathcal{V}_i(z_i)} } },
\end{multline}
where we define the commutator in the time-ordered product
(time-ordered with respect to the second argument) as
\begin{equation}
  [j(x),\mathcal{A}_m(y_m) ] \coloneq
  \Theta(y_{\alpha_0} - x_0)j(x)\mathcal{A}_m(y_m) -
  \Theta( x_0 - y_{\beta_0})\mathcal{A}_m(y_m)j(x);
\end{equation}
\begin{equation}
  \mathcal{T}\prod_{i=1}^n{\hat{\mathcal{A}}_i} =
  \cdots \hat{\mathcal{A}}_{\alpha}(y_\alpha) \hat{\mathcal{A}}_m(y_m)
  \hat{\mathcal{A}}_{\beta}(y_\beta) \cdots,
\end{equation}
using the Heaviside function $\Theta$. \textcolor{red}{See appendix for proof}.
The Noether current is only conserved on-shell so we cannot apply
current conservation to the second term (the path integral
is taken over all field configurations). Assuming that
the symmetry corresponding to $j^\mu$ leaves the path integral
measure invariant and transforms $\mathcal{V}_i$ as
\begin{equation}
  \mathcal{V}_i(\phi) \mapsto \mathcal{V}_i(\phi) + \delta_\epsilon\mathcal{V}_i(\phi)
  + \mathcal{O}(\epsilon^2); \quad \delta_\epsilon\mathcal{V}_i =
  \frac{\partial\mathcal{V}_i}{\partial\phi}\delta_\epsilon\phi,
\end{equation}
using the Dyson-Schwinger equation with $F[\phi] = \prod_i{\mathcal{V}_i}$
and~\eqref{eq:6} we have to first-order in $\epsilon$
\begin{equation}
  \Braket{\int{d^4x\,\epsilon\partial^x_\mu j^\mu(x)}\prod^n_{i=1}{\mathcal{V}_i(z_i)}} =
  -i\mathlarger{\mathlarger{\sum}}_{k =1}^n{\Braket{\delta_\epsilon\mathcal{V}_k(z_k)
  \prod_{i \neq k}^n{\mathcal{V}_i(z_i)}}}.
\end{equation}
Demanding the generator $\lambda(\phi, x)$ is smooth and compactly
supported and interchanging the spacetime and path integrals
leads to\footnote{n.b.\,\eqref{eq:1}$ \implies \partial^x_\mu j^\mu = -\lambda E$}
\begin{equation}
  \Braket{ \partial^x_\mu j^\mu(x)\prod^n_{i=1}{\mathcal{V}_i(z_i)} } = 
  -i\mathlarger{\mathlarger{\sum}}_{k=1}^n{\delta^{(4)}(x - z_k)
    \Braket{ \lambda(\phi, x)\frac{\partial\mathcal{V}_k}{\partial\phi}(z_k)
  \prod_{i \neq k}^n{\mathcal{V}_i(z_i)}} },
\end{equation}
via the fundamental lemma of the calculus of variations.
The delta function terms are called Schwinger/contact terms 
and in the case that none of the insertion times coincide we have 
the Ward-Takahashi identity~\cite{WardWTI,TakahashiWTI}
\begin{equation}
  \partial^x_{\mu}\!\Braket{j^\mu(x)\prod^n_{i=1}{\mathcal{V}_i(z_i)}} = 0;
  \quad x_0 \neq z_{i_0}.
\end{equation}
These correlators can be computed perturbatively, and in the case
of the axial current $j^\mu_5$, the three-point function corresponding
to triangle Feynman diagrams violates the Ward-Takahashi identity;
the Adler–Bell–Jackiw anomaly~\cite{Adler,Bell-Jackiw}.
This suggests that the axial symmetry is not present in the quantum theory.
Similarly, the Slavnov–Taylor identities imply it is a symmetry of
the quantum theory if the path integral Jacobian is non-singular
because the axial transformation is linear in the dirac fields.
Thus for the class of symmetries linear in the dynamical fields
in~\eqref{eq:7} a singular path integral Jacobian
characterises the presence of an anomaly.

\section{Anomalies in QFT}

\subsection{The Path Integral measure \& Fujikawa's method}

We shall now study transformations which are not necessarily symmetries
of the path integral measure, to this end we must introduce a more
precise notion of the path integral for fermionic fields using
Berezin's formal integration of Grassmann variables~\cite{BerezinBook}.
We decompose the Dirac spinor and its adjoint using an orthonormal basis $\{ \phi_n(x) \}$ of
the Dirac operator's eigenfunctions and the independent Grassmann variables 
$\theta_n$ and $\bar{\xi}_m$
\begin{equation}
  \psi = \sum_n{\theta_n\phi_n(x)} = \sum_n{\theta_n\!\braket{x|n}},
\end{equation}
\begin{equation}
  \bar{\psi} = \sum_m{\bar{\xi}_m\phi^{\dagger}_m(x)} = \sum_m{\bar{\xi}_m\!\braket{m|x}}.
  \end{equation}
Hence the path integral `measure' can be seen as a
Berezin integral `measure' under a coordinate transformation
\begin{equation}
  \theta \mapsto \theta_n\braket{x|n},\quad
  \bar{\xi} \mapsto \bar{\xi}_m\braket{m|x},
\end{equation}
\begin{equation}
  \mathcal{D}\psi\mathcal{D}\bar{\psi}
  = {\left[\det{\braket{x|n}}\det{\braket{m|x}}\right]}^{-1}
  \lim_{\substack{N \rightarrow \infty \\
  M \rightarrow \infty}}{\left\{ \prod_n^N{d\theta_n}\prod_m^M{d\bar{\xi}_m} \right\}}
  = \lim_{N \rightarrow \infty}{\prod_n^N{d\theta_n d\bar{\xi}_n}}.
\end{equation}
Under the infinitesimal vector transformation the Dirac spinor becomes
\begin{equation}
  \psi' = \sum_n{\theta_n'\phi_n(x)} ;\quad \theta_n' = e^{i\vartheta(x)}\theta_n,
\end{equation}
and similarly for the adjoint spinor. Thus the path integral measure transforms as
\begin{equation}
  \mathcal{D}\psi\mathcal{D}\bar{\psi} \mapsto
  {\left[\det{\left(e^{i\vartheta(x)\mathbb{I}}\right)}
  \det{\left(e^{-i\vartheta(x)\mathbb{I}}\right)} \right]}^{-1}
  \mathcal{D}\psi\mathcal{D}\bar{\psi},
\end{equation}
and we see the measure is invariant; the vector symmetry is retained
by the quantum theory and the correlators are gauge invariant
by the Slavnov–Taylor identities.
However, this is not the case for the axial transformation; 
the axial symmetry is `anomalous'. The following is an
outline of Fujikawa's method for the existence of the axial anomaly~\cite{Fujikawa}.
Under the axial transformation where the Grassmann variables transform as
\begin{equation}
  \theta_n' = \sum_m{C_{nm} \theta_m}, \quad \bar{\xi}_n' = \sum_m{C_{nm} \bar{\xi}_m},
\end{equation}
\begin{equation}
  C_{nm} = \delta_{nm} +i\int{d^4x\, \vartheta(x)\phi_{n}^{\dagger}(x)\gamma^5\phi_m(x)}.
\end{equation}
Using the identities
\begin{equation}
  \ln{(\mathbb{I} + \epsilon A)} = \epsilon A + \mathcal{O}(\epsilon^2),
\end{equation}
\begin{equation}
  \det\left( e^{A}\right) = e^{\mathrm{Tr}(A)},
\end{equation}
we find that the path integral measure changes by the Jacobian
\begin{equation}
  J[\vartheta] = {(\det{C})}^{-2} = \exp\left[-2 \mathrm{Tr}(\ln{C}) \right],
\end{equation}
\begin{equation}
  J[\vartheta] = \exp{\left[-2\mathrm{Tr}\left(i
  \int{d^4x\,  \vartheta(x)\phi_{n}^{\dagger}(x)\gamma^5\phi_m(x)} \right) \right]},
\end{equation}
\begin{equation}\label{eq:10}
  J[\vartheta] = \exp{\left[ -2i
  \int{d^4x\,\vartheta(x) \sum_n{\phi_{n}^{\dagger}(x)\gamma^5\phi_n(x)}} \right]}.
\end{equation}
The sum appearing in the Jacobian is clearly divergent (by orthonormality)
and requires the introduction of a regulator or ultraviolet cut-off to obtain a
non-singular Jacobian. However, there is a much more elegant approach available involving 
the Atiyah–Singer index theorem but first we show the relation between the axial anomaly
and the axial current. Henceforth, we define the axial anomaly $\mathcal{A}$
via the Jacobian of the path integral measure
\begin{equation}
   J[\vartheta] = \exp{\left[ -\int{d^4x\,\vartheta(x) \mathcal{A} } \right]},
\end{equation}
under the axial transformation the partition function becomes
\begin{equation}
  Z'[A_\mu, \vartheta] = \int{\mathcal{D}\psi'\mathcal{D}\bar{\psi}'\,
  \exp{\left[ \int{d^4x\,\bar{\psi}'(i\slashed{D} -m)\psi' -\frac{1}{4}F_{\mu\nu}F^{\mu\nu} } \right]}},
\end{equation}
but this is just a relabeling of the integration variables and thus the partition
function is invariant. Considering the change in the QED lagrangian we have
\begin{equation}
  Z[A_\mu, \vartheta] = \int{\mathcal{D}\psi\mathcal{D}\bar{\psi}\,J[\vartheta]
    \exp{ \left[ \int{d^4x\, \mathcal{L}_{\mathrm{QED}} + \vartheta(x)(\partial_\mu j^\mu_5
- 2im\bar{\psi}\gamma^5\psi) }  \right] } }.
\end{equation}
Thus to first order in $\vartheta(x)$ we have
\begin{equation}
   Z[A_\mu, \vartheta] = \int{\mathcal{D}\psi\mathcal{D}\bar{\psi}\, \left\{1 + \int{d^4x\,
     \vartheta(x)(\partial_\mu j^\mu_5 - 2im\bar{\psi}\gamma^5\psi -\mathcal{A})} \right\}
   \exp{\left[ \int{d^4x\, \mathcal{L}_{\mathrm{QED}} } \right]}},
\end{equation}
by the fundamental lemma of the calculus of variations~\cite{calcvar} we attain 
the anomalous divergence for the axial current
\begin{equation}
  \partial_\mu j^\mu_5 = 2im\bar{\psi}\gamma^5\psi + \mathcal{A}.
\end{equation}


\subsection{Atiyah–Singer index Theorem}

Given an eigenfunction of the Dirac operator
$\slashed{D}$, $\phi_n$, with an eigenvalue $\lambda_n \neq 0$ 
,$\gamma^5\phi_n$ is also an eigenfunction
\begin{equation}
  \slashed{D}(\gamma^5\phi_n) = -\gamma^5\slashed{D}(\phi_n) = -\lambda_n\gamma^5\phi_n,
\end{equation}
and the two are orthogonal as they have different eigenvalues.
However, when the eigenvalue is zero $\phi_n$ and $\gamma^5\phi_n$ correspond to $\lambda_n = 0$
and are called zero modes of $\slashed{D}$. We also have that $\gamma^5$ is 
hermitian thus by the spectral theorem there is a basis that diagonalises $\gamma^5$
and spans $\mathrm{Ker}(\slashed{D})$. The eigenvalues of $\gamma^5$ are $\pm 1$
(because ${(\gamma^5)}^2 = I_4$) and we use the projection
operators $P_\pm = \frac{1}{2}(I_4 \pm \gamma^5)$ to decompose
spinors into their positive and negative eigenvalue (positive and negative chirality respectively)
components. Ignoring the infinitesimal parameter for now, only the zero modes contribute
to the integral in the Jacobian because of orthonormality and we have
\begin{equation}
  \int{d^4x\, \sum_n{\phi^\dagger_n\gamma^5\phi_n}}
  = \sum_n{\int{d^4x\,\phi^{0\dagger}_{n+}\phi^0_{n+}}}
  -\sum_n{\int{d^4x\,\phi^{0\dagger}_{n-}\phi^0_{n-}}} = n_+ - n_-,
\end{equation}
\begin{equation}
  \int{d^4x\, \mathcal{A}} = 2i(n_+ - n_-),
\end{equation}
where $n_\pm$ are the number of positive and negative chirality zero modes $\phi^0_{n\pm}$
respectively. This is precisely the analytical index of the Dirac operator 
projected onto the positive chirality subspace
\begin{equation}
  \mathrm{index}(\slashed{D}_{+}) = n_+ - n_-, 
\end{equation}
\begin{equation}
  \slashed{D}_\pm \coloneq \slashed{D}P_{\pm} = \slashed{D}\big|_{\{\pm\}}.
\end{equation}
By the Atiyah–Singer index theorem this coincides with the Dirac operator's 
topological index on a compact manifold~\cite{Atiyah-Singer}.
On a 4-dimensional compact manifold $\Omega$ without boundary and
Riemannian curvature $R$ this index is~\cite{AST-Alvarez-Gaume,DiffGeomGauge}
\begin{equation}
  \mathrm{index}(\slashed{D}_+ ) =  \int_{\Omega}{\hat{A}(\Omega)\mathrm{ch}(F)},
\end{equation}
where the integrand, called the index density, only includes 4-form terms and the curvature
form is given by $F = e\,dA = \frac{e}{2}F_{\mu\nu}dx^\mu \wedge dx^\nu$.
The characteristic classes
$\mathrm{ch}(F)$ and $\hat{A}(\Omega)$ are the Chern character
of the curvature and Dirac genus of the manifold respectively
\begin{equation}
  \mathrm{ch}(F) = \mathrm{Tr}\left(\exp{\left[ \frac{i}{2\pi}F \right]}\right),
\end{equation}
\begin{equation}
  \hat{A}(\Omega) = \sqrt{ \det{\left( \frac{ \frac{i}{4\pi}R }{ \sinh{\frac{i}{4\pi}R} } \right) } }. 
\end{equation}
Now we pick the manifold $S^{4} = \Omega$ so the Riemannian curvature is 
constant and $\hat{A}(\Omega) = 1$. Thus the index becomes
\begin{equation}
  \mathrm{index}(\slashed{D}_+ ) =  \int_{S^4}{\mathrm{ch}(F)}
  = \frac{1}{2!}{\left(\frac{i}{2\pi}\right)}^2 \int_{S^4}{\mathrm{Tr}(F^2)}.
\end{equation}
Assuming that the action due to the Maxwell Lagrangian is finite
\begin{equation}\label{eq:8}
  \int_{S^4}{d^4x\, F_{\mu\nu}F^{\mu\nu}} < \infty \iff 
  A_\mu\big|_{S^4} \xrightarrow{x \rightarrow \infty} - \frac{i}{e}g^{-1}\partial_{\mu}g\big|_{S^4};
  \quad g(x)\in \mathrm{U}(1),
\end{equation}
we can map our problem from $S^4$ to $\mathbb{R}^4$
and $\mathrm{ch}(F)$ coincides with the Dirac operator's
index density in Euclidean space if the gauge fields $A_\mu$
obey~\eqref{eq:8}~\cite{JackiwRebbi1977,Belavin}.
Using antisymmetry of the wedge product we identify the
axial anomaly in Euclidean space as 
\begin{equation}
  \partial_\mu j^\mu_5 - 2im\bar{\psi}\gamma^5\psi =
  \mathcal{A}[A_\mu] = \frac{-ie^2}{16\pi^2}
  \mathrm{Tr}(\epsilon^{\mu\nu\rho\sigma} F_{\mu\nu}F_{\rho\sigma}).
\end{equation}

\subsection{Anomalies as obstructions to gauging}

We have seen that a singular path integral Jacobian is a necessary and 
sufficient condition for anomalous linear symmetries.
In the general case, gauging such a symmetry will
produce a gauge-variant partition function because constructing a gauge invariant
classical action ensures the shift in the measure changes the partition function.
This gauge symmetry breaking is tantamount to a violation of unitarity\footnote{i.e probability
non-conservation} and is fatal for any quantum theory.

We now study the effect of gauging an anomalous symmetries in the case of
the Dirac Lagrangian's axial symmetry. Using Fujikawa's method~\cite{Fujikawa}
we shall calculate the anomaly for the `axial theory' although a topological approach analogous to
the previous section is possible using the Atiyah–Singer Families
index theorem~\cite{AlvarezGuameNon-AB,Sumitani1984,FamiliesIndex}. For the theory of massless
Dirac fermions coupling to a $\mathrm{U}(1)$ axial boson in~\eqref{eq:9}, 
the path integral in Euclidean space is 
\begin{equation}
  \int{\mathcal{D}\psi\mathcal{D}\bar{\psi}\,
    \exp{\left[- \int{d^4x\, \bar{\psi}i\widetilde{\slashed{D}}\psi
  -\frac{1}{4}\mathrm{Tr}(K_{\mu\nu}K^{\mu\nu})} \right]}}.
\end{equation}
The classical action is gauge-invariant by construction, thus the only change in
the path integral under a gauge transformation is from the Jacobian
\begin{equation}
    \int{\mathcal{D}\psi'\mathcal{D}\bar{\psi}'\, e^{-S[\psi',\bar{\psi}']} } =
    \int{\mathcal{D}\psi\mathcal{D}\bar{\psi}\, J[\vartheta] e^{-S[\psi,\bar{\psi}]} },
\end{equation}
where the Jacobian takes the same form as in~\eqref{eq:10} for an orthonormal
basis of eigenfunctions $\{ \phi_n(x)\}$ of the operator $\widetilde{\slashed{D}}$.
We now proceed via Fujikawa's method of gauge-invariant mode cut-off
to calculate the Jacobian by introducing a smooth regulator $f(x)$ with
all derivatives vanishing at infinity
\begin{equation}
  J[\vartheta] = \exp{\left[ -2i \lim_{N \rightarrow \infty}{ \left\{
  \sum_{n=1}^N{ \int{d^4x\, \vartheta(x)\phi_n^\dagger\gamma^5\phi_n} } \right\} } \right]}, 
\end{equation}
\begin{equation}
  J[\vartheta] = \exp{\left[ -2i \lim_{M \rightarrow \infty}{ \left\{
          \sum_{n=1}^{\infty}{ \int{d^4x\, \vartheta(x)\phi_n^\dagger\gamma^5
          f\left(\frac{\lambda_n^2}{M^2}\right) \phi_n} } \right\} } \right]}, 
\end{equation}
\begin{equation}
  J[\vartheta] = \exp{\left[ -2i \lim_{M \rightarrow \infty}{
  \mathrm{Tr}\left\{ \vartheta(x)\gamma^5
f\left( \frac{ {\widetilde{\slashed{D}}}^2 }{M^2}\right) \right\}} \right]}, 
\end{equation}
where the functional trace is taken over spacetime using the basis $\{ \phi_n(x) \}$.
Now we compute the series using momentum space
\begin{equation}
  \phi_n(x) = \int{\frac{d^4k}{(2\pi)^2}\,\tilde{\phi}_n(k)e^{ikx}   },
\end{equation}
\begin{equation}
  \sum_{n = 1}^{\infty}{\phi_n^\dagger\gamma^5
  f\left( \frac{\widetilde{\slashed{D}}{}^2 }{M^2} \right)\phi_n} =
  \int{\frac{d^4k d^4p}{(2\pi)^4} \sum_{n=1}^{\infty}{ e^{-ipx}\tilde{\phi}_n^\dagger(p)
      \gamma^5 f\left( \frac{ \widetilde{\slashed{D}}{}^2 }{M^2}\right) \tilde{\phi}_n(k)
  e^{ikx} } },
\end{equation}
\begin{equation}
  =  \int{\frac{d^4k}{(2\pi)^4}\, \mathrm{Tr}\left\{
      \gamma^5 f\left( \frac{(\widetilde{D}_\mu +ik_\mu)(\widetilde{D}^\mu +ik^\mu)}{M^2}
        + \frac{iq [\gamma^\mu,\gamma^\nu ]K_{\mu\nu} }{4M^2} \right)  \right\} }, 
\end{equation}
where we have used the identities
\begin{equation}
  \sum_{n = 1}^{\infty}{\tilde{\phi}_n^\dagger(p) A \tilde{\phi}_n(k)} =
  \delta(k-p) \mathrm{Tr}(A),
\end{equation}
\begin{equation}
  \widetilde{\slashed{D}}{}^2 = \widetilde{D}_\mu \widetilde{D}^\mu
  + \frac{iq}{4}[\gamma^\mu, \gamma^\nu]K_{\mu\nu},
\end{equation}
\begin{equation}
  e^{-ikx}f(\partial_\mu)e^{ikx} = f(\partial_\mu +ik_\mu).
\end{equation}
Now we rescale the momenta via $k_\mu \rightarrow Mk_\mu$ and the series becomes
\begin{equation}
  M^4 \int{\frac{d^4k}{(2\pi)^4}\, \mathrm{Tr}\left\{
      \gamma^5 f\left( \frac{ \widetilde{D}_\mu \widetilde{D}^\mu
        + 2iMk_\mu \widetilde{D}^\mu}{M^2} -k^2
        + \frac{iq [\gamma^\mu,\gamma^\nu ]K_{\mu\nu} }{4M^2} \right)  \right\} }, 
\end{equation}
expanding the Taylor series of $f(x)$ about $-k^2 = - k_\mu k^\mu$, and using the trace 
properties of the gamma matrices the only term non-vanishing in
the limit $M \rightarrow \infty$ is the term quadratic in $\Gamma_{\mu\nu}$ 
\begin{equation}
  \frac{M^4q^2}{2!M^4}\mathrm{Tr}(\epsilon^{\mu\nu\rho\sigma}K_{\mu\nu}K_{\rho\sigma})
  \int{\frac{d^4k}{(2\pi)^4}f''(-k^2) },
\end{equation}
and we have 
\begin{equation}
J[\vartheta] = \exp{\left[ \int{d^4x\, \frac{-iq^2}{16\pi^2}
      \mathrm{Tr}(\epsilon^{\mu\nu\rho\sigma}K_{\mu\nu}K_{\rho\sigma})
      \vartheta(x)} \right]},
\end{equation}
thus there is gauge-variance in the path integral measure and the quantum theory
is inconsistent.
